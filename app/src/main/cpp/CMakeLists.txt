cmake_minimum_required(VERSION 3.22.1)
project(readflow_teleocr LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)

get_filename_component(READFLOW_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(LLAMA_SOURCE "${READFLOW_ROOT}/.deps/teleocr-llama")
if(NOT EXISTS "${LLAMA_SOURCE}/tools/mtmd/mtmd.h")
    message(FATAL_ERROR "Run python3 scripts/prepare_teleocr.py before building ReadFlow.")
endif()
execute_process(COMMAND git -C "${LLAMA_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE LLAMA_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE LLAMA_GIT_STATUS)
if(NOT LLAMA_GIT_STATUS EQUAL 0 OR NOT LLAMA_REVISION STREQUAL "2280f0fd163b03d4331d49e08460f66502e35854")
    message(FATAL_ERROR "Wrong llama.cpp revision. Run python3 scripts/prepare_teleocr.py.")
endif()
file(READ "${LLAMA_SOURCE}/src/models/qwen2vl.cpp" TELEOCR_QWEN_SOURCE)
if(NOT TELEOCR_QWEN_SOURCE MATCHES "layer.attn_q_norm = create_tensor")
    message(FATAL_ERROR "Missing TeleOCR architecture patch. Run python3 scripts/prepare_teleocr.py.")
endif()
add_subdirectory("${LLAMA_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
add_library(readflow_teleocr SHARED teleocr_jni.cpp)
target_link_libraries(readflow_teleocr PRIVATE llama mtmd)
# Export only JNI entry points; all llama/ggml symbols remain private to this library.
target_link_options(readflow_teleocr PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-z,max-page-size=16384")
